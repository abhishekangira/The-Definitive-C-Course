# Pointers are addresses

```c id=pointer-addresses run
#include <stdio.h>

int main(void) {
    int a = 1, b = 2;
    int *p = &a;
    int *q = NULL;
    printf("&a = %p\n", (void *)&a);
    printf("&b = %p\n", (void *)&b);
    printf("p  = %p (same as &a? %s)\n", (void *)p, p == &a ? "yes" : "no");
    printf("q  = %p\n", (void *)q);
    printf("hex literal stays: %#x\n", 255u);
    return 0;
}
```

```c id=heap-overflow run sanitize=address,undefined expect=runtime-error
#include <stdlib.h>

int main(void) {
    int *a = malloc(4 * sizeof *a);
    if (a == NULL) {
        return 1;
    }
    a[4] = 7;
    free(a);
    return 0;
}
```
