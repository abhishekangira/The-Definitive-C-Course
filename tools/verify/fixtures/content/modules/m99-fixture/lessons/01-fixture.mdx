Standard input arrives through a pipe.

```c id=fx-sum-stdin stdin="3 4\n"
#include <stdio.h>

int main(void) {
    int a, b;

```

Pointers print differently everywhere, so only `%p` output is normalised. The `%#x` stays.

```c id=fx-pointers
#include <stdio.h>

int main(void) {
    int x = 42;
    int y = 7;
    int *p = &x;
    int *nothing = NULL;
    printf("&x      = %p\n", (void *)&x);
    printf("&y      = %p\n", (void *)&y);
    printf("p       = %p (same as &x? %s)\n", (void *)p, p == &x ? "yes" : "no");
    printf("nothing = %p\n", (void *)nothing);
    printf("magic   = %#x\n", 0xdeadbeefu);
    return 0;
}
```

The stepper shows `swap` changing `x` and `y` through pointers.

```c id=fx-swap trace
#include <stdio.h>

void swap(int *a, int *b) {
    int tmp = *a;
    *a = *b;
    *b = tmp;
}

int main(void) {
    int x = 1;
    int y = 2;
    swap(&x, &y);
    printf("x=%d y=%d\n", x, y);
    return 0;
}
```

Buffering: a terminal flushes each line, a pipe does not, so a crash loses unflushed pipe output.

```c id=fx-buffering stdout=both expect=runtime-error predict=outcome
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    printf("this line was printed\n");
    abort();
}
```

A native-only POSIX sample.

```c id=fx-fork target=native posix
#define _POSIX_C_SOURCE 200809L
#include <stdio.h>
#include <sys/wait.h>
#include <unistd.h>

int main(void) {
    fflush(stdout);
    pid_t pid = fork();
    if (pid == 0) {
        printf("child\n");
        return 3;
    }
    int status = 0;
    waitpid(pid, &status, 0);
    printf("child exited with %d\n", WEXITSTATUS(status));
    return 0;
}
```
